   800→export function DemoTemplateClient({ template }: { template: string }) {
   801→  const store = demoStores[template]
   802→  const products = demoProducts[template] || []
   803→  const [selectedProductId, setSelectedProductId] = useState<string | null>(null)
   804→
   805→  if (!store) return null
   806→
   807→  const planId = templatePlanId[template] || 'free'
   808→  const isPremium = planId === 'premium'
   809→  const isFree = planId === 'free'
   810→  const planLabel = getPlanLabel(template)
   811→
   812→  // Populate Zustand store so ProductDetailView can find the data
   813→  const setStoreData = () => {
   814→    useAppStore.setState({
   815→      stores: [store],
   816→      products: products,
   817→      currentStore: store,
   818→    })
   819→  }
   820→
   821→  // Initialize Zustand on mount (useEffect instead of render-time mutation)
   822→  useEffect(() => {
   823→    setStoreData()
   824→  }, [])
   825→
   826→  // Handle product click from templates
   827→  const handleProductClick = (productId: string) => {
   828→    setStoreData()
   829→    setSelectedProductId(productId)
   830→    window.scrollTo({ top: 0, behavior: 'smooth' })
   831→  }
   832→
   833→  // Handle back from product detail
   834→  const handleBackToStore = () => {
   835→    setSelectedProductId(null)
   836→    window.scrollTo({ top: 0, behavior: 'smooth' })
   837→  }
   838→
   839→  // Banner style based on plan
   840→  const bannerStyle = isPremium

   842→    : isFree
   843→    ? 'bg-emerald-600 text-white'
   844→    : 'bg-violet-600 text-white'
   845→
   846→  const BannerIcon = isPremium ? Crown : isFree ? Gem : Sparkles

   851→      <div className="relative">
   852→        {/* Demo Banner */}

   856→              <BannerIcon className="w-3.5 h-3.5" />

   858→            </div>
   859→          </div>

   868→  return (
   869→    <div className="relative">

   890→          >
