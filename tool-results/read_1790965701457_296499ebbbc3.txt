   418→export function DemoTemplateClient({ template }: { template: string }) {
   419→  const store = demoStores[template]
   420→  const products = demoProducts[template] || []
   421→  const [selectedProductId, setSelectedProductId] = useState<string | null>(null)
   422→
   423→  if (!store) return null
   424→
   425→  const planId = templatePlanId[template] || 'free'
   426→  const isPremium = planId === 'premium'
   427→  const isFree = planId === 'free'
   428→  const planLabel = getPlanLabel(template)
   429→
   430→  // Populate Zustand store so ProductDetailView can find the data
   431→  const setStoreData = () => {
   432→    useAppStore.setState({
   433→      stores: [store],
   434→      products: products,
   435→      currentStore: store,
   436→    })
   437→  }
   438→
   439→  // Initialize Zustand on mount (useEffect instead of render-time mutation)
   440→  useEffect(() => {
   441→    setStoreData()

   443→
   444→  // Handle product click from templates
   445→  const handleProductClick = (productId: string) => {
   446→    setStoreData()

   448→    window.scrollTo({ top: 0, behavior: 'smooth' })
   449→  }
   450→
   451→  // Handle back from product detail

   456→
   457→  // Banner style based on plan

   462→    : 'bg-violet-600 text-white'
